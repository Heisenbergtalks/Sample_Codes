import pandas as pd
import numpy as np
import statsmodels.api as sm
from scipy.stats import poisson
import matplotlib.pyplot as plt
from matplotlib.patches import Patch
import warnings
warnings.filterwarnings("ignore")

# ── LOAD ALL DATA ─────────────────────────────────────────────────────────
master = pd.read_csv("master_atlantic_features_1984_2025.csv")
cmip6  = pd.read_csv("cmip6_daily.csv")

# ── CMIP6 ENVIRONMENT INDEX ───────────────────────────────────────────────
cmip6["date"]      = pd.to_datetime(cmip6["time_index"])
cmip6["year"]      = cmip6["date"].dt.year
cmip6["month"]     = cmip6["date"].dt.month
cmip6["precip_mm"] = cmip6["precip"] * 86400

season = cmip6[cmip6["month"].between(6,11)].copy()

baseline    = season[season["year"].between(2015,2025)]
base_temp   = baseline.groupby("year")["max_temp"].mean().mean()
base_precip = baseline.groupby("year")["precip_mm"].sum().mean()
base_rh     = baseline.groupby("year")[
    "relativehumidity_2m"].mean().mean()

print(f"CMIP6 baselines (2015-2025):")
print(f"  Temp:   {base_temp:.2f} C")
print(f"  Precip: {base_precip:.1f} mm")
print(f"  RH:     {base_rh:.1f} %")

cmip6_annual = []
for yr in range(2015, 2060):
    yr_data = season[season["year"]==yr]
    if len(yr_data) < 100:
        continue
    aug_oct     = yr_data[yr_data["month"].between(8,10)]
    temp_anom   = yr_data["max_temp"].mean() - base_temp
    precip_anom = yr_data["precip_mm"].sum() - base_precip
    rh_mean     = aug_oct["relativehumidity_2m"].mean()
    cmip6_annual.append({
        "year":        yr,
        "temp_anom":   round(temp_anom, 3),
        "precip_anom": round(precip_anom, 1),
        "rh_aug_oct":  round(rh_mean, 2),
    })

cmip6_df = pd.DataFrame(cmip6_annual)

# Normalize for combined index
for col in ["temp_anom","precip_anom","rh_aug_oct"]:
    mn = cmip6_df[col].min()
    mx = cmip6_df[col].max()
    cmip6_df[f"{col}_norm"] = (cmip6_df[col]-mn)/(mx-mn)

cmip6_df["env_index"] = (
    cmip6_df["temp_anom_norm"] +
    cmip6_df["precip_anom_norm"] +
    cmip6_df["rh_aug_oct_norm"]
) / 3

print(f"\nCMIP6 environment index:")
print(f"  Min: {cmip6_df['env_index'].min():.3f}")
print(f"  Max: {cmip6_df['env_index'].max():.3f}")
print(f"  2015-2025 mean: "
      f"{cmip6_df[cmip6_df['year']<=2025]['env_index'].mean():.3f}")
print(f"  2046-2059 mean: "
      f"{cmip6_df[cmip6_df['year']>=2046]['env_index'].mean():.3f}")

# ── MERGE ─────────────────────────────────────────────────────────────────
df = master.merge(
    cmip6_df[["year","temp_anom","precip_anom",
              "rh_aug_oct","env_index"]],
    on="year", how="left")

# env_index filled with 0.5 for pre-2015
# Raw CMIP6 components kept as NaN pre-2015
df["env_index"] = df["env_index"].fillna(0.5)

print(f"\nMerged: {len(df)} rows")
print(f"CMIP6 overlap: "
      f"{(~df['env_index'].eq(0.5)).sum()} years")

# ── TRAIN / VALIDATION SPLIT ──────────────────────────────────────────────
train = df[df["year"] <= 2015].copy()
valid = df[df["year"] >  2015].copy()
y_train = train["n_storms_within_150mi"]
y_valid = valid["n_storms_within_150mi"]

print(f"\nTrain: {len(train)} years (1984-2015)")
print(f"Valid: {len(valid)} years (2016-2025)")
print(f"Train hits: {y_train.sum()}  "
      f"mean: {y_train.mean():.3f}")
print(f"Valid hits: {y_valid.sum()}  "
      f"mean: {y_valid.mean():.3f}")

# ── FIT FOUR MODELS ───────────────────────────────────────────────────────
results = {}

# Model A: ENSO + ACE (baseline)
Xa_tr = sm.add_constant(train[["enso_aso","ace_index"]])
Xa_va = sm.add_constant(valid[["enso_aso","ace_index"]])
ma    = sm.Poisson(y_train, Xa_tr).fit(disp=False)
pa    = ma.predict(Xa_va)
results["A: ENSO + ACE"] = {
    "model":ma, "pred":pa,
    "mae":abs(y_valid-pa).mean(),
    "aic":ma.aic}

# Model B: ENSO + ACE + env_index
Xb_tr = sm.add_constant(
    train[["enso_aso","ace_index","env_index"]])
Xb_va = sm.add_constant(
    valid[["enso_aso","ace_index","env_index"]])
mb    = sm.Poisson(y_train, Xb_tr).fit(disp=False)
pb    = mb.predict(Xb_va)
results["B: ENSO + ACE + CMIP6 index"] = {
    "model":mb, "pred":pb,
    "mae":abs(y_valid-pb).mean(),
    "aic":mb.aic}

# Model C: ENSO + env_index only
Xc_tr = sm.add_constant(
    train[["enso_aso","env_index"]])
Xc_va = sm.add_constant(
    valid[["enso_aso","env_index"]])
mc    = sm.Poisson(y_train, Xc_tr).fit(disp=False)
pc    = mc.predict(Xc_va)
results["C: ENSO + CMIP6 index"] = {
    "model":mc, "pred":pc,
    "mae":abs(y_valid-pc).mean(),
    "aic":mc.aic}

# Model D: ENSO + ACE + raw CMIP6 components
# Only train on years with CMIP6 data
train_d = train.dropna(subset=[
    "temp_anom","precip_anom","rh_aug_oct"])
valid_d = valid.dropna(subset=[
    "temp_anom","precip_anom","rh_aug_oct"])

print(f"\nModel D training rows (with CMIP6): {len(train_d)}")

if len(train_d) >= 3:
    Xd_tr = sm.add_constant(
        train_d[["enso_aso","ace_index",
                 "temp_anom","precip_anom",
                 "rh_aug_oct"]])
    Xd_va = sm.add_constant(
        valid_d[["enso_aso","ace_index",
                 "temp_anom","precip_anom",
                 "rh_aug_oct"]])
    md = sm.Poisson(
        train_d["n_storms_within_150mi"],
        Xd_tr).fit(disp=False)
    pd_ = md.predict(Xd_va)
    results["D: ENSO+ACE+CMIP6 raw"] = {
        "model":md, "pred":pd_,
        "mae":abs(valid_d["n_storms_within_150mi"]-pd_).mean(),
        "aic":md.aic}
else:
    print("Model D skipped — insufficient overlap")

# ── MODEL COMPARISON ──────────────────────────────────────────────────────
print("\n" + "="*65)
print("MODEL COMPARISON")
print("="*65)
baseline_mae = results["A: ENSO + ACE"]["mae"]
print(f"\n{'Model':<35} {'AIC':>8} {'Val MAE':>10} {'vs Baseline':>12}")
print("-"*68)
for name, res in results.items():
    diff   = baseline_mae - res["mae"]
    better = (f"+{diff:.3f} better" if diff > 0
              else "BASELINE" if diff == 0
              else f"{diff:.3f} worse")
    print(f"{name:<35} {res['aic']:>8.1f} "
          f"{res['mae']:>10.3f} {better:>12}")

# ── BEST MODEL ────────────────────────────────────────────────────────────
best_name = min(results, key=lambda x: results[x]["mae"])
best      = results[best_name]
best_m    = best["model"]
best_p    = best["pred"]

print(f"\nWINNER: {best_name}")
print(f"  AIC:     {best['aic']:.1f}")
print(f"  Val MAE: {best['mae']:.3f}")
print(f"  Improvement: "
      f"+{baseline_mae-best['mae']:.3f} storms/year")

print("\n" + "="*65)
print(f"BEST MODEL COEFFICIENTS")
print("="*65)
print(best_m.summary().tables[1])

# ── VALIDATION DETAIL ─────────────────────────────────────────────────────
print(f"\nValidation year by year (2016-2025):")
print(f"{'Year':<6} {'Actual':>8} {'Predicted':>10} "
      f"{'Error':>8} {'In 95% CI':>10}")
print("-"*48)

# Use matching valid set
if "D" in best_name:
    val_use = valid_d
    pred_use = best_p
else:
    val_use  = valid
    pred_use = best_p

for yr, act, pred in zip(
    val_use["year"], 
    val_use["n_storms_within_150mi"],
    pred_use):
    lo  = poisson.ppf(0.025, max(pred,0.001))
    hi  = poisson.ppf(0.975, max(pred,0.001))
    err = act - pred
    ci  = "✓" if lo <= act <= hi else "✗"
    print(f"{int(yr):<6} {act:>8.0f} {pred:>10.2f} "
          f"{err:>+8.2f} {ci:>10}")

# ── FULL HISTORICAL FIT ───────────────────────────────────────────────────
feat_cols = list(best_m.params.index[1:])
X_all     = sm.add_constant(df[feat_cols])
df["fitted"] = best_m.predict(X_all)

# ── PROJECTION 2026-2059 ──────────────────────────────────────────────────
enso_markov = pd.read_csv(
    "enso_projection_markov_2025_2059.csv")
ace_mean = df["ace_index"].mean()
ace_std  = df["ace_index"].std()
enso_std = df["enso_aso"].std()

np.random.seed(42)
n_sims     = 10000
proj_years = list(range(2026, 2060))
year_results = {}

for yr in proj_years:
    enso_row     = enso_markov[
        enso_markov["year"]==yr].iloc[0]
    enso_central = enso_row["enso_aso_central"]

    cmip6_yr = cmip6_df[cmip6_df["year"]==yr]
    if len(cmip6_yr) > 0:
        env_val    = cmip6_yr["env_index"].values[0]
        temp_val   = cmip6_yr["temp_anom"].values[0]
        precip_val = cmip6_yr["precip_anom"].values[0]
        rh_val     = cmip6_yr["rh_aug_oct"].values[0]
    else:
        env_val    = 0.5
        temp_val   = 0.0
        precip_val = 0.0
        rh_val     = base_rh

    sim_counts = []
    for _ in range(n_sims):
        enso_sim = np.random.normal(enso_central, enso_std)
        ace_sim  = max(0, np.random.normal(ace_mean, ace_std))

        if best_name == "A: ENSO + ACE":
            X_sim = pd.DataFrame({
                "const":     [1.0],
                "enso_aso":  [enso_sim],
                "ace_index": [ace_sim]})
        elif best_name == "B: ENSO + ACE + CMIP6 index":
            X_sim = pd.DataFrame({
                "const":     [1.0],
                "enso_aso":  [enso_sim],
                "ace_index": [ace_sim],
                "env_index": [env_val]})
        elif best_name == "C: ENSO + CMIP6 index":
            X_sim = pd.DataFrame({
                "const":     [1.0],
                "enso_aso":  [enso_sim],
                "env_index": [env_val]})
        else:
            X_sim = pd.DataFrame({
                "const":      [1.0],
                "enso_aso":   [enso_sim],
                "ace_index":  [ace_sim],
                "temp_anom":  [temp_val],
                "precip_anom":[precip_val],
                "rh_aug_oct": [rh_val]})

        lam   = best_m.predict(X_sim)[0]
        count = np.random.poisson(max(lam,0.001))
        sim_counts.append(count)

    sim_arr = np.array(sim_counts)
    year_results[yr] = {
        "year":         yr,
        "enso_central": round(enso_central, 2),
        "env_index":    round(env_val, 3),
        "expected":     round(float(np.mean(sim_arr)), 3),
        "p10":          int(np.percentile(sim_arr, 10)),
        "p25":          int(np.percentile(sim_arr, 25)),
        "p50":          int(np.percentile(sim_arr, 50)),
        "p75":          int(np.percentile(sim_arr, 75)),
        "p90":          int(np.percentile(sim_arr, 90)),
        "p_zero":       round(float(np.mean(sim_arr==0)), 3),
        "p_one":        round(float(np.mean(sim_arr==1)), 3),
        "p_two":        round(float(np.mean(sim_arr==2)), 3),
        "p_three_plus": round(float(np.mean(sim_arr>=3)), 3),
        "risk_level":   ("HIGH"     if enso_central < -0.5
                         else "LOW" if enso_central >  0.5
                         else "MODERATE")
    }

proj_df = pd.DataFrame(list(year_results.values()))
proj_df.to_csv(
    "final_projection_2026_2059.csv", index=False)

# ── PRINT PROJECTIONS ─────────────────────────────────────────────────────
print("\n" + "="*65)
print("YEAR BY YEAR PROJECTIONS 2026-2059")
print("="*65)
print(f"{'Year':<6} {'ENSO':>6} {'Risk':>10} "
      f"{'Expected':>10} {'P(0)':>8} "
      f"{'P(1)':>8} {'P(2+)':>8}")
print("-"*60)
for _, r in proj_df.iterrows():
    p2p = r["p_two"] + r["p_three_plus"]
    print(f"{int(r['year']):<6} "
          f"{r['enso_central']:>+6.2f} "
          f"{r['risk_level']:>10} "
          f"{r['expected']:>10.2f} "
          f"{r['p_zero']:>7.1%} "
          f"{r['p_one']:>8.1%} "
          f"{p2p:>8.1%}")

hist_mean = df["n_storms_within_150mi"].mean()
avg_lam   = proj_df[
    proj_df["year"].between(2026,2035)
]["expected"].mean()

print(f"\nDecade summaries:")
for period in ["2026-2035","2036-2045","2046-2059"]:
    y1,y2  = int(period[:4]),int(period[5:])
    yr_exp = proj_df[
        proj_df["year"].between(y1,y2)
    ]["expected"].values
    sim_dec = np.array([
        sum(np.random.poisson(max(e,0.001))
            for e in yr_exp)
        for _ in range(n_sims)
    ])
    exp = yr_exp.sum()
    lo  = int(np.percentile(sim_dec,20))
    hi  = int(np.percentile(sim_dec,80))
    print(f"  {period}: {exp:.1f} expected "
          f"({lo}-{hi}, 60% CI)")
print(f"  Historical/decade: {hist_mean*10:.1f}")

print(f"\nCumulative P(at least 1 storm):")
for n in [1,2,3,5,10]:
    prob = 1 - np.exp(-avg_lam*n)
    print(f"  Within {n:2d} year(s): {prob:.1%}")
