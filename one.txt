import pandas as pd
import numpy as np

# ── Load CMIP6 ────────────────────────────────────────────────────────────
cmip6 = pd.read_csv("cmip6_daily.csv")

# ── Parse date ────────────────────────────────────────────────────────────
cmip6["date"]  = pd.to_datetime(cmip6["time_index"])
cmip6["year"]  = cmip6["date"].dt.year
cmip6["month"] = cmip6["date"].dt.month

# ── Unit conversions ──────────────────────────────────────────────────────
cmip6["precip_mm"] = cmip6["precip"] * 86400
cmip6["tmax_C"]    = cmip6["max_temp"]
cmip6["tavg_C"]    = cmip6["temperature_2m"]
cmip6["wind_ms"]   = cmip6["wnd_spd_10m"]

# ── Filter to hurricane season Jun-Nov ────────────────────────────────────
season = cmip6[cmip6["month"].between(6, 11)].copy()

# ── CMIP6 internal baseline 2015-2025 ────────────────────────────────────
baseline_yr   = season[season["year"].between(2015, 2025)]
baseline_prcp = baseline_yr.groupby("year")["precip_mm"].sum().mean()
baseline_tmax = baseline_yr.groupby("year")["tmax_C"].mean().mean()
baseline_wind = baseline_yr.groupby("year")["wind_ms"].mean().mean()

print("CMIP6 internal baseline (2015-2025):")
print(f"  precip: {baseline_prcp:.1f} mm")
print(f"  tmax:   {baseline_tmax:.2f} C")
print(f"  wind:   {baseline_wind:.2f} m/s")

# ── Count columns definition ──────────────────────────────────────────────
count_cols = [
    "count_days_25mm",
    "count_days_50mm",
    "count_days_75mm",
    "longest_dry_spell",
    "count_days_90f",
    "count_days_95f",
    "count_wind_5ms",
    "count_wind_7ms",
    "compound_days",
    "count_rh_80",
]

# ── Engineer annual features ──────────────────────────────────────────────
rows = []

for yr in range(2015, 2060):
    yr_data = season[season["year"] == yr].copy()

    if len(yr_data) < 100:
        print(f"WARNING: {yr} only {len(yr_data)} days — skipping")
        continue

    aug_oct = yr_data[yr_data["month"].between(8, 10)]

    # ── PRECIPITATION ─────────────────────────────────────────────────────
    total_prcp      = yr_data["precip_mm"].sum()
    max_1day_prcp   = yr_data["precip_mm"].max()
    max_7day_prcp   = yr_data["precip_mm"].rolling(
                        7,  min_periods=7).sum().max()
    max_14day_prcp  = yr_data["precip_mm"].rolling(
                        14, min_periods=14).sum().max()
    count_days_25mm = int((yr_data["precip_mm"] > 25.4).sum())
    count_days_50mm = int((yr_data["precip_mm"] > 50.8).sum())
    count_days_75mm = int((yr_data["precip_mm"] > 76.2).sum())

    # Anomaly computed from CMIP6 internal baseline
    # NOT from GHCND — keeps it self consistent
    prcp_anomaly    = total_prcp - baseline_prcp

    # Longest dry spell
    is_dry          = (yr_data["precip_mm"] < 2.54).astype(int)
    dry_groups      = is_dry.groupby(
                        (is_dry != is_dry.shift()).cumsum())
    longest_dry     = int(dry_groups.sum().max()) if len(is_dry) > 0 else 0

    # ── TEMPERATURE ───────────────────────────────────────────────────────
    mean_tmax         = yr_data["tmax_C"].mean()
    tmax_anomaly      = mean_tmax - baseline_tmax
    mean_tmax_aug_oct = aug_oct["tmax_C"].mean()
    mean_tavg_aug_oct = aug_oct["tavg_C"].mean()
    count_days_90f    = int((yr_data["tmax_C"] > 32.2).sum())
    count_days_95f    = int((yr_data["tmax_C"] > 35.0).sum())

    # ── WIND ──────────────────────────────────────────────────────────────
    mean_wind      = yr_data["wind_ms"].mean()
    max_wind       = yr_data["wind_ms"].max()
    count_wind_5ms = int((yr_data["wind_ms"] > 5.0).sum())
    count_wind_7ms = int((yr_data["wind_ms"] > 7.0).sum())
    awnd_anomaly   = mean_wind - baseline_wind

    # ── HUMIDITY ──────────────────────────────────────────────────────────
    mean_rh_aug_oct = aug_oct["relativehumidity_2m"].mean()
    count_rh_80     = int((yr_data["relativehumidity_2m"] > 80).sum())

    # ── COMPOUND ──────────────────────────────────────────────────────────
    compound_days = int((
        (yr_data["precip_mm"] > 25.4) &
        (yr_data["wind_ms"]   > 5.0)
    ).sum())

    # ── WIND DIRECTION ────────────────────────────────────────────────────
    mean_u10_aug_oct = aug_oct["u10"].mean()
    mean_v10_aug_oct = aug_oct["v10"].mean()

    rows.append({
        "year":               yr,
        "total_prcp_mm":      round(total_prcp, 1),
        "max_1day_prcp_mm":   round(max_1day_prcp, 1),
        "max_7day_prcp_mm":   round(max_7day_prcp, 1),
        "max_14day_prcp_mm":  round(max_14day_prcp, 1),
        "count_days_25mm":    count_days_25mm,
        "count_days_50mm":    count_days_50mm,
        "count_days_75mm":    count_days_75mm,
        "prcp_anomaly_mm":    round(prcp_anomaly, 1),
        "longest_dry_spell":  longest_dry,
        "mean_tmax_C":        round(mean_tmax, 2),
        "tmax_anomaly_C":     round(tmax_anomaly, 2),
        "mean_tmax_aug_oct":  round(mean_tmax_aug_oct, 2),
        "mean_tavg_aug_oct":  round(mean_tavg_aug_oct, 2),
        "count_days_90f":     count_days_90f,
        "count_days_95f":     count_days_95f,
        "mean_awnd_ms":       round(mean_wind, 2),
        "max_awnd_ms":        round(max_wind, 2),
        "count_wind_5ms":     count_wind_5ms,
        "count_wind_7ms":     count_wind_7ms,
        "awnd_anomaly_ms":    round(awnd_anomaly, 3),
        "max_wsf2_ms":        None,
        "max_wsf5_ms":        None,
        "count_wsf2_10ms":    None,
        "compound_days":      compound_days,
        "mean_rh_aug_oct":    round(mean_rh_aug_oct, 2),
        "count_rh_80":        count_rh_80,
        "mean_u10_aug_oct":   round(mean_u10_aug_oct, 3),
        "mean_v10_aug_oct":   round(mean_v10_aug_oct, 3),
    })

df = pd.DataFrame(rows)
print(f"\nRaw CMIP6 features: {len(df)} years ({df['year'].min()}-{df['year'].max()})")

# ── Bias correction using 2015-2025 overlap ───────────────────────────────
ghcnd = pd.read_csv("ghcnd_features_annual_1984_2025.csv")

ghcnd_overlap = ghcnd[ghcnd["year"].between(2015, 2025)]
cmip6_overlap = df[df["year"].between(2015, 2025)]

# Features to bias correct
# Only continuous features — counts already integers
bc_features = [
    "total_prcp_mm",
    "max_1day_prcp_mm",
    "max_7day_prcp_mm",
    "max_14day_prcp_mm",
    "mean_tmax_C",
    "tmax_anomaly_C",
    "mean_tmax_aug_oct",
    "mean_tavg_aug_oct",
    "mean_awnd_ms",
    "max_awnd_ms",
    "awnd_anomaly_ms",
]

print("\nBias correction table:")
print(f"{'Feature':<25} {'GHCND':>10} {'CMIP6':>10} {'Bias':>10}")
print("-" * 60)

biases = {}
for feat in bc_features:
    if feat in ghcnd_overlap.columns and feat in cmip6_overlap.columns:
        g_mean = ghcnd_overlap[feat].mean()
        c_mean = cmip6_overlap[feat].mean()
        bias   = g_mean - c_mean
        biases[feat] = bias
        print(f"{feat:<25} {g_mean:>10.2f} {c_mean:>10.2f} {bias:>10.2f}")

# Apply bias correction to continuous features only
df_corrected = df.copy()
for feat, bias in biases.items():
    df_corrected[feat] = (df_corrected[feat] + bias).round(2)

# Recompute prcp_anomaly from CMIP6 internal baseline
# This keeps anomaly self-consistent within CMIP6
cmip6_baseline_prcp = df[
    df["year"].between(2015, 2025)
]["total_prcp_mm"].mean()

df_corrected["prcp_anomaly_mm"] = (
    df_corrected["total_prcp_mm"] - cmip6_baseline_prcp
).round(1)

print(f"\nCMIP6 internal precip baseline: {cmip6_baseline_prcp:.1f}mm")
print("prcp_anomaly_mm recomputed from CMIP6 internal baseline")

# ── Save ──────────────────────────────────────────────────────────────────
df.to_csv(
    "cmip6_features_raw_2015_2059.csv",
    index=False)

df_corrected.to_csv(
    "cmip6_features_bias_corrected_2015_2059.csv",
    index=False)

print(f"\nSaved: cmip6_features_raw_2015_2059.csv")
print(f"Saved: cmip6_features_bias_corrected_2015_2059.csv")

# ── Sanity check ──────────────────────────────────────────────────────────
print("\nSanity check — overlap period 2015-2025:")
print(f"{'Year':<6} {'GHCND_prcp':>12} {'CMIP6_raw':>12} {'CMIP6_corr':>12} {'compound':>10}")
print("-" * 55)

ghcnd_dict = ghcnd_overlap.set_index("year")["total_prcp_mm"].to_dict()
raw_dict   = df[df["year"].between(2015,2025)].set_index("year")["total_prcp_mm"].to_dict()
corr_dict  = df_corrected[df_corrected["year"].between(2015,2025)].set_index("year")["total_prcp_mm"].to_dict()
comp_dict  = df_corrected[df_corrected["year"].between(2015,2025)].set_index("year")["compound_days"].to_dict()

for yr in range(2015, 2026):
    g  = ghcnd_dict.get(yr, "—")
    r  = raw_dict.get(yr, "—")
    c  = corr_dict.get(yr, "—")
    cd = comp_dict.get(yr, "—")
    print(f"{yr:<6} {str(g):>12} {str(r):>12} {str(c):>12} {str(cd):>10}")

# ── Count column check ────────────────────────────────────────────────────
print("\nCount column check (should all be integers):")
for col in count_cols:
    if col in df_corrected.columns:
        sample = df_corrected[col].head(3).tolist()
        print(f"  {col}: {sample}")
