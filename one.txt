import pandas as pd
import numpy as np
from scipy import stats
from scipy.stats import pointbiserialr, poisson
import matplotlib.pyplot as plt
import warnings
warnings.filterwarnings("ignore")

# ── LOAD DATA ─────────────────────────────────────────────────────────────
ghcnd  = pd.read_csv("ghcnd_mco_1984_2025.csv")
cmip6  = pd.read_csv("cmip6_daily.csv")
hits   = pd.read_csv("hurdat2_annual_hit_table_1984_2025.csv")

# ── GHCND PROCESSING ──────────────────────────────────────────────────────
ghcnd["DATE"]   = pd.to_datetime(ghcnd["DATE"])
ghcnd["year"]   = ghcnd["DATE"].dt.year
ghcnd["month"]  = ghcnd["DATE"].dt.month
ghcnd["TMAX_C"] = (ghcnd["TMAX"] - 32) * 5/9
ghcnd["TMIN_C"] = (ghcnd["TMIN"] - 32) * 5/9
ghcnd["TAVG_C"] = (ghcnd["TAVG"] - 32) * 5/9

# Annual aggregates
ghcnd_annual = ghcnd.groupby("year").agg(
    annual_max_temp_C  = ("TMAX_C", "max"),
    mean_tmax_C        = ("TMAX_C", "mean"),
    mean_tmin_C        = ("TMIN_C", "mean"),
    mean_tavg_C        = ("TAVG_C", "mean"),
).reset_index()

# Hurricane season Jun-Nov
season = ghcnd[ghcnd["month"].between(6,11)]
season_annual = season.groupby("year").agg(
    season_mean_tmax_C = ("TMAX_C", "mean"),
    season_max_tmax_C  = ("TMAX_C", "max"),
).reset_index()

ghcnd_annual = ghcnd_annual.merge(
    season_annual, on="year", how="left")

print("GHCND annual data:")
print(f"  Years: {ghcnd_annual['year'].min()}"
      f"-{ghcnd_annual['year'].max()}")
print(f"  Mean TMAX range: "
      f"{ghcnd_annual['mean_tmax_C'].min():.1f}"
      f"-{ghcnd_annual['mean_tmax_C'].max():.1f} C")

# ── CMIP6 PROCESSING ──────────────────────────────────────────────────────
cmip6["date"]  = pd.to_datetime(cmip6["time_index"])
cmip6["year"]  = cmip6["date"].dt.year
cmip6["month"] = cmip6["date"].dt.month

cmip6_annual = cmip6.groupby("year").agg(
    annual_max_temp_C  = ("max_temp", "max"),
    mean_tmax_C        = ("max_temp", "mean"),
).reset_index()

cmip6_season = cmip6[cmip6["month"].between(6,11)]
cmip6_season_annual = cmip6_season.groupby("year").agg(
    season_mean_tmax_C = ("max_temp", "mean"),
    season_max_tmax_C  = ("max_temp", "max"),
).reset_index()

cmip6_annual = cmip6_annual.merge(
    cmip6_season_annual, on="year", how="left")

# ── BIAS CORRECTION ───────────────────────────────────────────────────────
overlap_ghcnd = ghcnd_annual[
    ghcnd_annual["year"].between(2015,2025)]
overlap_cmip6 = cmip6_annual[
    cmip6_annual["year"].between(2015,2025)]

bias_mean   = (overlap_ghcnd["mean_tmax_C"].mean() -
               overlap_cmip6["mean_tmax_C"].mean())
bias_max    = (overlap_ghcnd["annual_max_temp_C"].mean() -
               overlap_cmip6["annual_max_temp_C"].mean())
bias_season = (overlap_ghcnd["season_mean_tmax_C"].mean() -
               overlap_cmip6["season_mean_tmax_C"].mean())

print(f"\nBias corrections:")
print(f"  Mean tmax:   {bias_mean:+.2f} C")
print(f"  Annual max:  {bias_max:+.2f} C")
print(f"  Season mean: {bias_season:+.2f} C")

cmip6_annual["mean_tmax_C"]        += bias_mean
cmip6_annual["annual_max_temp_C"]  += bias_max
cmip6_annual["season_mean_tmax_C"] += bias_season

cmip6_proj = cmip6_annual[
    cmip6_annual["year"] >= 2026].copy()

print(f"\nCMIP6 projection:")
print(f"  2026 mean tmax: "
      f"{cmip6_proj[cmip6_proj['year']==2026]['mean_tmax_C'].values[0]:.2f} C")
print(f"  2059 mean tmax: "
      f"{cmip6_proj[cmip6_proj['year']==2059]['mean_tmax_C'].values[0]:.2f} C")

# ── MERGE HITS ────────────────────────────────────────────────────────────
hits = hits[["year","hit"]]
hits["year"] = hits["year"].astype(int)
hits["hit"]  = hits["hit"].astype(int)

ghcnd_annual["year"] = ghcnd_annual["year"].astype(int)

df = ghcnd_annual.merge(hits, on="year", how="left")
df = df[df["year"] <= 2025].copy()
df = df.dropna(subset=["hit","mean_tmax_C"])
df["hit"] = df["hit"].astype(int)

print(f"\nMaster dataset: {len(df)} years")
print(f"Hit years:   {df['hit'].sum()}")
print(f"Quiet years: {(df['hit']==0).sum()}")

# ── LINEAR TREND ──────────────────────────────────────────────────────────
hist_years = df["year"].values
hist_temps = df["mean_tmax_C"].values

slope, intercept, r, p, se = stats.linregress(
    hist_years, hist_temps)

print(f"\nLinear trend (annual mean TMAX):")
print(f"  Slope:      {slope:.4f} C/year")
print(f"  Per decade: {slope*10:.3f} C/decade")
print(f"  R²:         {r**2:.3f}")
print(f"  P-value:    {p:.4f}")
print(f"  Significant: {'YES' if p < 0.05 else 'NO'}")

# Full trend line 1984-2059
all_years  = np.arange(1984, 2060)
trend_line = intercept + slope * all_years

# 95% confidence intervals
n      = len(hist_years)
x_mean = hist_years.mean()
se_line = se * np.sqrt(
    1/n + (all_years - x_mean)**2 /
    np.sum((hist_years - x_mean)**2))

ci_upper = trend_line + 1.96 * se_line
ci_lower = trend_line - 1.96 * se_line

# ── TEMPERATURE HURRICANE CORRELATION ─────────────────────────────────────
baseline_temp      = df["mean_tmax_C"].mean()
df["temp_anomaly"] = df["mean_tmax_C"] - baseline_temp

corr, pval = pointbiserialr(
    df["hit"], df["temp_anomaly"])

warm_years = df[df["temp_anomaly"] > 0]
cool_years = df[df["temp_anomaly"] <= 0]
warm_hit   = warm_years["hit"].mean()
cool_hit   = cool_years["hit"].mean()

print(f"\nTemperature-Hurricane relationship:")
print(f"  Baseline temp:      {baseline_temp:.2f} C")
print(f"  Correlation:        {corr:.3f}")
print(f"  P-value:            {pval:.3f}")
print(f"  Significant:        {'YES' if pval < 0.05 else 'NO'}")
print(f"  Warm year hit rate: {warm_hit:.1%}")
print(f"  Cool year hit rate: {cool_hit:.1%}")
print(f"  Difference:         {(warm_hit-cool_hit)*100:.1f}%")

# ── PROJECT HURRICANE PROBABILITY ─────────────────────────────────────────
hist_hit_rate = df["hit"].mean()

print(f"\nHistorical base rate: {hist_hit_rate:.3f} per year")
print(f"Historical per decade: {hist_hit_rate*10:.1f} storms")

proj_rows = []
for yr in range(2026, 2060):

    # Use trend line for temperature projection
    # More stable than CMIP6 raw values
    proj_temp = intercept + slope * yr

    temp_above = proj_temp - baseline_temp

    # IPCC AR6 Chapter 11:
    # +1C warming = +5% hurricane frequency
    adj_factor = 1 + (temp_above * 0.05)
    proj_prob  = min(hist_hit_rate * adj_factor, 0.95)

    # Poisson lambda
    lam = -np.log(1 - proj_prob)

    proj_rows.append({
        "year":           yr,
        "proj_temp_C":    round(proj_temp, 2),
        "temp_above_C":   round(temp_above, 2),
        "adj_factor":     round(adj_factor, 3),
        "proj_prob":      round(proj_prob, 3),
        "poisson_lambda": round(lam, 3),
    })

proj_df = pd.DataFrame(proj_rows)
proj_df.to_csv(
    "temperature_projection_2026_2059.csv",
    index=False)

print("\nProjected annual probability:")
for yr in [2026,2030,2035,2040,2050,2059]:
    row = proj_df[proj_df["year"]==yr].iloc[0]
    print(f"  {yr}: temp={row['proj_temp_C']:.2f}C  "
          f"prob={row['proj_prob']:.1%}  "
          f"λ={row['poisson_lambda']:.3f}")

# ── YEAR BY YEAR TABLE ────────────────────────────────────────────────────
print("\n" + "="*60)
print("YEAR BY YEAR PROJECTIONS 2026-2059")
print("="*60)
print(f"{'Year':<6} {'Temp(C)':>8} {'Ann Prob':>10} "
      f"{'λ/year':>8} {'Cumul Prob':>12}")
print("-"*50)

avg_lam = proj_df[
    proj_df["year"].between(2026,2035)
]["poisson_lambda"].mean()

for _, row in proj_df.iterrows():
    yr   = int(row["year"])
    temp = row["proj_temp_C"]
    prob = row["proj_prob"]
    lam  = row["poisson_lambda"]
    yrs_from_now = yr - 2025
    cumul = 1 - np.exp(-avg_lam * yrs_from_now)
    print(f"{yr:<6} {temp:>8.2f} {prob:>9.1%} "
          f"{lam:>8.3f} {cumul:>11.1%}")

# ── PER YEAR STORM COUNT PROBABILITIES ───────────────────────────────────
print("\n" + "="*60)
print("ANNUAL STORM COUNT PROBABILITIES BY YEAR")
print("="*60)
print(f"{'Year':<6} {'λ':>6} {'P(0 storms)':>13} "
      f"{'P(1 storm)':>12} {'P(2 storms)':>12} "
      f"{'P(3+ storms)':>13}")
print("-"*65)

for yr in [2026,2028,2030,2032,2035,
           2038,2040,2042,2045,
           2050,2055,2059]:
    row = proj_df[proj_df["year"]==yr].iloc[0]
    lam = row["poisson_lambda"]
    p0  = poisson.pmf(0, lam)
    p1  = poisson.pmf(1, lam)
    p2  = poisson.pmf(2, lam)
    p3  = 1 - poisson.cdf(2, lam)
    print(f"{yr:<6} {lam:>6.3f} {p0:>12.1%} "
          f"{p1:>12.1%} {p2:>12.1%} {p3:>12.1%}")

print()
print("P(0) = no storms hit Orlando that year")
print("P(1) = exactly 1 storm hits that year")
print("P(2) = exactly 2 storms hit that year")
print("P(3+)= 3 or more storms hit that year")

# Historical comparison
hist_lam = -np.log(1 - hist_hit_rate)
h0 = poisson.pmf(0, hist_lam)
h1 = poisson.pmf(1, hist_lam)
h2 = poisson.pmf(2, hist_lam)
h3 = 1 - poisson.cdf(2, hist_lam)
print(f"\nHistorical baseline (1984-2025):")
print(f"  λ = {hist_lam:.3f}")
print(f"  P(0 storms): {h0:.1%}")
print(f"  P(1 storm):  {h1:.1%}")
print(f"  P(2 storms): {h2:.1%}")
print(f"  P(3+ storms):{h3:.1%}")

# ── MONTE CARLO ───────────────────────────────────────────────────────────
np.random.seed(42)
n_sims = 10000

print("\n" + "="*60)
print("DECADE SUMMARY — EXPECTED STORM COUNTS")
print("="*60)
print(f"{'Period':<12} {'Expected':>10} "
      f"{'Low(20%)':>10} {'High(80%)':>10}")
print("-"*45)

decade_data = {
    "Historical":  None,
    "2026-2035":   proj_df[proj_df["year"].between(2026,2035)],
    "2036-2045":   proj_df[proj_df["year"].between(2036,2045)],
    "2046-2059":   proj_df[proj_df["year"].between(2046,2059)],
}

mc_results = {}
for period, data in decade_data.items():
    if period == "Historical":
        sim_counts = np.random.poisson(
            hist_lam, (n_sims,10)).sum(axis=1)
        expected   = round(hist_hit_rate * 10, 1)
    else:
        lambdas    = data["poisson_lambda"].values
        sim_counts = np.array([
            sum(np.random.poisson(l) for l in lambdas)
            for _ in range(n_sims)
        ])
        expected = round(lambdas.sum(), 1)

    low  = int(np.percentile(sim_counts, 20))
    high = int(np.percentile(sim_counts, 80))

    mc_results[period] = {
        "expected": expected,
        "low": low,
        "high": high
    }
    print(f"{period:<12} {expected:>10} "
          f"{low:>10} {high:>10}")

# ── CUMULATIVE PROBABILITY ────────────────────────────────────────────────
print(f"\nCumulative P(at least 1 hurricane impact):")
print(f"  Near term λ = {avg_lam:.3f}/year")
print()
for n in [1,2,3,5,10,15,20]:
    prob = 1 - np.exp(-avg_lam * n)
    print(f"  Within {n:2d} year(s):  {prob:.1%}")

# ── MAIN CHART ────────────────────────────────────────────────────────────
fig, ax = plt.subplots(figsize=(16,8))

ax.fill_between(all_years, ci_lower, ci_upper,
    alpha=0.12, color="#E74C3C",
    label="95% confidence band")

ax.plot(df["year"], df["mean_tmax_C"],
    color="#95A5A6", linewidth=1.2,

    label="Observed annual mean max temp (GHCND)")

ax.plot(cmip6_proj["year"],

    color="#E67E22", linewidth=1.5,
    linestyle="--", alpha=0.8, zorder=2,
    label="CMIP6 projected temp (EC-Earth3-Veg SSP2-4.5)")

hist_mask = all_years <= 2025
ax.plot(all_years[hist_mask],
    trend_line[hist_mask],
    color="#E74C3C", linewidth=2.5,
    zorder=3,
    label=f"Linear trend "
          f"(+{slope*10:.3f}°C/decade, "
          f"R²={r**2:.2f}, p={p:.4f})")

proj_mask = all_years >= 2025
ax.plot(all_years[proj_mask],
    trend_line[proj_mask],
    color="#E74C3C", linewidth=2.5,
    linestyle="--", zorder=3)

hit_df   = df[df["hit"]==1]
quiet_df = df[df["hit"]==0]

ax.scatter(hit_df["year"],
    hit_df["mean_tmax_C"],
    color="#1B2A4A", s=150,
    zorder=5, marker="^",
    edgecolors="white", linewidth=0.8,
    label=f"Hurricane impact year "
          f"(n={len(hit_df)}, "
          f"{warm_hit:.0%} warm / "
          f"{cool_hit:.0%} cool yr hit rate)")

ax.scatter(quiet_df["year"],
    quiet_df["mean_tmax_C"],
    color="#85C1E9", s=60,
    zorder=4, marker="o",
    alpha=0.7, edgecolors="white",
    linewidth=0.5,
    label=f"Quiet year (n={len(quiet_df)})")

ax.axvline(x=2025.5, color="black",
    linewidth=1.5, linestyle=":",
    alpha=0.6, zorder=3)
ax.text(2026, df["mean_tmax_C"].min()+0.1,
    "Today →", fontsize=9,
    color="black", alpha=0.7)

for _, row in hit_df.iterrows():
    if row["year"] in [2004,2017,2022,2024]:
        ax.annotate(
            str(int(row["year"])),
            xy=(row["year"],
                row["mean_tmax_C"]),
            xytext=(row["year"]+0.5,
                    row["mean_tmax_C"]+0.1),
            fontsize=7, color="#1B2A4A",
            arrowprops=dict(
                arrowstyle="-",
                color="#1B2A4A",
                lw=0.8))

ax.set_xlabel("Year", fontsize=12)
ax.set_ylabel(
    "Annual Mean Max Temperature (°C)",
    fontsize=12)
ax.set_title(
    "Orlando MCO — Temperature Trend & "
    "Hurricane Impact Years  |  1984–2059\n"
    "GHCND Observed (1984–2025) + "
    "EC-Earth3-Veg CMIP6 Projection "
    "(2026–2059)  |  "
    "▲ = Hurricane impact  ● = Quiet year",
    fontsize=11, pad=15)

ax.legend(loc="upper left", fontsize=8,
    framealpha=0.9)
ax.grid(True, alpha=0.25)
ax.set_xlim(1982, 2061)

plt.tight_layout()
plt.savefig("temperature_hurricane_trend.png",
    dpi=200, bbox_inches="tight")
plt.show()
print("\nSaved: temperature_hurricane_trend.png")
