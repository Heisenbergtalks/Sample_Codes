import pandas as pd
import numpy as np

# ── Load CMIP6 ────────────────────────────────────────────────────────────
cmip6 = pd.read_csv("cmip6_daily.csv")

# ── Parse date ────────────────────────────────────────────────────────────
cmip6["date"] = pd.to_datetime(cmip6["time_index"])
cmip6["year"]  = cmip6["date"].dt.year
cmip6["month"] = cmip6["date"].dt.month

# ── Unit conversions ──────────────────────────────────────────────────────
# precip: kg/m2/s to mm/day
cmip6["precip_mm"] = cmip6["precip"] * 86400

# temperature already in Celsius
cmip6["tmax_C"]    = cmip6["max_temp"]
cmip6["tavg_C"]    = cmip6["temperature_2m"]

# wind already in m/s
cmip6["wind_ms"]   = cmip6["wnd_spd_10m"]

# wind direction ratio from u10/v10
cmip6["wind_dir_ratio"] = cmip6["u10"] / (cmip6["v10"].replace(0, np.nan))

# ── Filter to hurricane season Jun-Nov ────────────────────────────────────
season = cmip6[cmip6["month"].between(6, 11)].copy()

# ── Baseline from 2015-2025 overlap period ────────────────────────────────
baseline = season[season["year"].between(2015, 2025)]
baseline_precip = baseline.groupby("year")["precip_mm"].sum().mean()
baseline_tmax   = baseline.groupby("year")["tmax_C"].mean().mean()
baseline_wind   = baseline.groupby("year")["wind_ms"].mean().mean()

print(f"CMIP6 baseline precip (Jun-Nov): {baseline_precip:.1f} mm")
print(f"CMIP6 baseline tmax  (Jun-Nov): {baseline_tmax:.2f} C")
print(f"CMIP6 baseline wind  (Jun-Nov): {baseline_wind:.2f} m/s")

# ── Engineer annual features ──────────────────────────────────────────────
rows = []

for yr in range(2015, 2060):
    yr_data = season[season["year"] == yr]

    if len(yr_data) < 100:
        print(f"WARNING: {yr} only {len(yr_data)} days")
        continue

    aug_oct = yr_data[yr_data["month"].between(8, 10)]

    # PRECIPITATION
    total_prcp        = yr_data["precip_mm"].sum()
    max_1day_prcp     = yr_data["precip_mm"].max()
    max_7day_prcp     = yr_data["precip_mm"].rolling(7,  min_periods=7).sum().max()
    max_14day_prcp    = yr_data["precip_mm"].rolling(14, min_periods=14).sum().max()
    count_days_25mm   = (yr_data["precip_mm"] > 25.4).sum()
    count_days_50mm   = (yr_data["precip_mm"] > 50.8).sum()
    count_days_75mm   = (yr_data["precip_mm"] > 76.2).sum()
    prcp_anomaly      = total_prcp - baseline_precip
    longest_dry       = (yr_data["precip_mm"] < 2.54
                        ).astype(int).groupby(
                        (yr_data["precip_mm"] >= 2.54)
                        .astype(int).cumsum()).sum().max()

    # TEMPERATURE
    mean_tmax         = yr_data["tmax_C"].mean()
    tmax_anomaly      = mean_tmax - baseline_tmax
    mean_tmax_aug_oct = aug_oct["tmax_C"].mean()
    mean_tavg_aug_oct = aug_oct["tavg_C"].mean()
    count_days_90f    = (yr_data["tmax_C"] > 32.2).sum()
    count_days_95f    = (yr_data["tmax_C"] > 35.0).sum()

    # WIND
    mean_wind         = yr_data["wind_ms"].mean()
    max_wind          = yr_data["wind_ms"].max()
    count_wind_5ms    = (yr_data["wind_ms"] > 5.0).sum()
    count_wind_7ms    = (yr_data["wind_ms"] > 7.0).sum()
    awnd_anomaly      = mean_wind - baseline_wind

    # HUMIDITY (bonus)
    mean_rh_aug_oct   = aug_oct["relativehumidity_2m"].mean()
    count_rh_80       = (yr_data["relativehumidity_2m"] > 80).sum()

    # COMPOUND (precip + wind threshold)
    compound_days     = (
        (yr_data["precip_mm"] > 25.4) &
        (yr_data["wind_ms"]   > 5.0)
    ).sum()

    # WIND DIRECTION
    mean_u10_aug_oct  = aug_oct["u10"].mean()
    mean_v10_aug_oct  = aug_oct["v10"].mean()

    rows.append({
        "year":               yr,
        "total_prcp_mm":      round(total_prcp, 1),
        "max_1day_prcp_mm":   round(max_1day_prcp, 1),
        "max_7day_prcp_mm":   round(max_7day_prcp, 1),
        "max_14day_prcp_mm":  round(max_14day_prcp, 1),
        "count_days_25mm":    int(count_days_25mm),
        "count_days_50mm":    int(count_days_50mm),
        "count_days_75mm":    int(count_days_75mm),
        "prcp_anomaly_mm":    round(prcp_anomaly, 1),
        "longest_dry_spell":  int(longest_dry),
        "mean_tmax_C":        round(mean_tmax, 2),
        "tmax_anomaly_C":     round(tmax_anomaly, 2),
        "mean_tmax_aug_oct":  round(mean_tmax_aug_oct, 2),
        "mean_tavg_aug_oct":  round(mean_tavg_aug_oct, 2),
        "count_days_90f":     int(count_days_90f),
        "count_days_95f":     int(count_days_95f),
        "mean_awnd_ms":       round(mean_wind, 2),
        "max_awnd_ms":        round(max_wind, 2),
        "count_wind_5ms":     int(count_wind_5ms),
        "count_wind_7ms":     int(count_wind_7ms),
        "awnd_anomaly_ms":    round(awnd_anomaly, 3),
        "max_wsf2_ms":        None,
        "max_wsf5_ms":        None,
        "count_wsf2_10ms":    None,
        "compound_days":      int(compound_days),
        "mean_rh_aug_oct":    round(mean_rh_aug_oct, 2),
        "count_rh_80":        int(count_rh_80),
        "mean_u10_aug_oct":   round(mean_u10_aug_oct, 3),
        "mean_v10_aug_oct":   round(mean_v10_aug_oct, 3),
    })

# ── Save raw CMIP6 features ───────────────────────────────────────────────
df = pd.DataFrame(rows)
df.to_csv("cmip6_features_raw_2015_2059.csv", index=False)
print(f"\nRaw CMIP6 features: {len(df)} years")

# ── Bias correction using 2015-2025 overlap ───────────────────────────────
# Load GHCND features for same period
ghcnd = pd.read_csv("ghcnd_features_annual_1984_2025.csv")
ghcnd_overlap = ghcnd[ghcnd["year"].between(2015, 2025)]
cmip6_overlap = df[df["year"].between(2015, 2025)]

# Features to bias correct
# Must match exactly between GHCND and CMIP6
bc_features = [
    "total_prcp_mm",
    "max_1day_prcp_mm",
    "max_7day_prcp_mm",
    "max_14day_prcp_mm",
    "count_days_25mm",
    "count_days_50mm",
    "count_days_75mm",
    "prcp_anomaly_mm",
    "longest_dry_spell",
    "mean_tmax_C",
    "tmax_anomaly_C",
    "mean_tmax_aug_oct",
    "mean_tavg_aug_oct",
    "count_days_90f",
    "count_days_95f",
    "mean_awnd_ms",
    "max_awnd_ms",
    "count_wind_5ms",
    "count_wind_7ms",
    "awnd_anomaly_ms",
    "compound_days",
]

print("\nBias correction (GHCND mean - CMIP6 mean):")
biases = {}
for feat in bc_features:
    if feat in ghcnd_overlap.columns and feat in cmip6_overlap.columns:
        ghcnd_mean = ghcnd_overlap[feat].mean()
        cmip6_mean = cmip6_overlap[feat].mean()
        bias       = ghcnd_mean - cmip6_mean
        biases[feat] = bias
        print(f"  {feat}: GHCND={ghcnd_mean:.2f} "
              f"CMIP6={cmip6_mean:.2f} bias={bias:.2f}")

# Apply bias to all CMIP6 years
df_corrected = df.copy()
for feat, bias in biases.items():
    df_corrected[feat] = df_corrected[feat] + bias

# Save bias corrected
df_corrected.to_csv("cmip6_features_bias_corrected_2015_2059.csv", index=False)
print(f"\nBias corrected CMIP6 features saved")
print(f"Years: {df_corrected['year'].min()} - {df_corrected['year'].max()}")
