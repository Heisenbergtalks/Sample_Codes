import pandas as pd
import numpy as np
from xgboost import XGBClassifier
from sklearn.metrics import (classification_report, 
                             confusion_matrix,
                             roc_auc_score)
import matplotlib.pyplot as plt

# ── Load data ────────────────────────────────────────────────────────────
df = pd.read_csv("master_feature_table_1984_2025.csv")

# ── Define features ───────────────────────────────────────────────────────
drop_cols = [
    "year",
    "hit",
    "n_storms_within_150mi",
    "closest_storm",
    "closest_dist_mi",
    "enso_phase",        # categorical — enso_aso covers this
]

feature_cols = [c for c in df.columns if c not in drop_cols]
print(f"Features used: {len(feature_cols)}")
print(feature_cols)

X = df[feature_cols]
y = df["hit"]

# ── Train / validation split ──────────────────────────────────────────────
# MUST be time based — no random splits
train = df[df["year"] <= 2015]
valid = df[df["year"] >  2015]

X_train = train[feature_cols]
y_train = train["hit"]
X_valid = valid[feature_cols]
y_valid = valid["hit"]

print(f"\nTrain: {len(train)} years (1984-2015)")
print(f"Valid: {len(valid)} years (2016-2025)")
print(f"Train hits: {y_train.sum()} / {len(y_train)}")
print(f"Valid hits: {y_valid.sum()} / {len(y_valid)}")

# ── Class imbalance weight ────────────────────────────────────────────────
n_neg = (y_train == 0).sum()
n_pos = (y_train == 1).sum()
scale = n_neg / n_pos
print(f"\nClass weight (scale_pos_weight): {scale:.2f}")

# ── Train XGBoost ─────────────────────────────────────────────────────────
model = XGBClassifier(
    n_estimators      = 100,
    max_depth         = 3,       # shallow — prevents overfit on small data
    learning_rate     = 0.05,    # slow learning
    scale_pos_weight  = scale,   # handle class imbalance
    subsample         = 0.8,     # row sampling
    colsample_bytree  = 0.8,     # feature sampling
    random_state      = 42,
    eval_metric       = "logloss",
    verbosity         = 0,
)

model.fit(
    X_train, y_train,
    eval_set=[(X_valid, y_valid)],
    verbose=False,
)

# ── Predictions ───────────────────────────────────────────────────────────
train_probs = model.predict_proba(X_train)[:, 1]
valid_probs = model.predict_proba(X_valid)[:, 1]
valid_preds = (valid_probs >= 0.5).astype(int)

# ── Validation results ────────────────────────────────────────────────────
print("\n" + "="*50)
print("VALIDATION RESULTS (2016-2025)")
print("="*50)

print("\nYear by year:")
for i, (yr, prob, actual) in enumerate(
    zip(valid["year"], valid_probs, y_valid)):
    flag = "✓" if (prob >= 0.5) == actual else "✗"
    print(f"  {yr}: P(hit)={prob:.2f}  actual={actual}  {flag}")

print(f"\nROC-AUC score: {roc_auc_score(y_valid, valid_probs):.3f}")
print(f"\nClassification report:")
print(classification_report(y_valid, valid_preds,
      target_names=["No hit","Hit"]))

print("\nConfusion matrix:")
cm = confusion_matrix(y_valid, valid_preds)
print(f"  True Quiet predicted Quiet:    {cm[0][0]}")
print(f"  True Quiet predicted Hit:      {cm[0][1]}")
print(f"  True Hit   predicted Quiet:    {cm[1][0]}")
print(f"  True Hit   predicted Hit:      {cm[1][1]}")

# ── Feature importance ────────────────────────────────────────────────────
print("\n" + "="*50)
print("FEATURE IMPORTANCE (top 10)")
print("="*50)

importance = pd.DataFrame({
    "feature":    feature_cols,
    "importance": model.feature_importances_,
}).sort_values("importance", ascending=False)

print(importance.head(10).to_string(index=False))

# ── Full probability output 1984-2025 ────────────────────────────────────
all_probs = model.predict_proba(X)[:, 1]
results = pd.DataFrame({
    "year":        df["year"].values,
    "hit_actual":  y.values,
    "p_hit":       all_probs.round(3),
    "predicted":   (all_probs >= 0.5).astype(int),
    "correct":     ((all_probs >= 0.5).astype(int) == y.values).astype(int),
    "split":       ["train" if yr <= 2015 else "valid"
                    for yr in df["year"].values],
})

results.to_csv("xgboost_annual_probabilities.csv", index=False)
print(f"\nSaved: xgboost_annual_probabilities.csv")

# ── Save model ────────────────────────────────────────────────────────────
model.save_model("xgboost_hurricane_model.json")
print("Saved: xgboost_hurricane_model.json")

# ── Plot feature importance ───────────────────────────────────────────────
top10 = importance.head(10)
plt.figure(figsize=(8,5))
plt.barh(top10["feature"][::-1], 
         top10["importance"][::-1])
plt.xlabel("Importance")
plt.title("XGBoost Feature Importance — Top 10")
plt.tight_layout()
plt.savefig("feature_importance.png", dpi=150)
print("Saved: feature_importance.png")
