import pandas as pd
import numpy as np
import xgboost as xgb
from sklearn.ensemble import RandomForestRegressor, GradientBoostingRegressor, ExtraTreesRegressor
from sklearn.metrics import mean_absolute_error
import copy, warnings
warnings.filterwarnings('ignore')

# ── Merge all features ─────────────────────────────
df = pd.read_csv('/content/florida_hurricane_training_data.csv')
df = df.sort_values('year').reset_index(drop=True)

# Add steering features
df = df.merge(steering_all, on='year', how='left')

# ── Full feature engineering ───────────────────────
df['prev1']           = df['hurricane_count'].shift(1)
df['prev2']           = df['hurricane_count'].shift(2)
df['prev3']           = df['hurricane_count'].shift(3)
df['prev_3yr_avg']    = df['hurricane_count'].rolling(3).mean().shift(1)
df['prev_5yr_avg']    = df['hurricane_count'].rolling(5).mean().shift(1)
df['prev_10yr_avg']   = df['hurricane_count'].rolling(10).mean().shift(1)
df['sst_anom']        = df['atlantic_sst'] - df['atlantic_sst'].mean()
df['gulf_anom']       = df['gulf_sst'] - df['gulf_sst'].mean()
df['sst_x_amo']       = df['atlantic_sst'] * df['amo']
df['sst_x_enso']      = df['atlantic_sst'] * (-df['enso'])
df['gulf_x_amo']      = df['gulf_sst'] * df['amo']
df['shear_x_enso']    = df['wind_shear'] * df['enso']
df['fav_index']       = (df['sst_anom']*2 + df['gulf_anom']*2 +
                         (-df['enso'])*1.5 + df['amo']*2 +
                         (-df['wind_shear']+12)*1)
df['post_1995']       = (df['year'] >= 1995).astype(int)
df['strong_lanina']   = (df['enso'] < -0.8).astype(int)
df['amo_positive']    = (df['amo'] > 0.1).astype(int)
df['rain_anom']       = df['orlando_rain'] - df['orlando_rain'].mean()
df['sst_trend_5yr']   = df['atlantic_sst'].rolling(5).mean().shift(1)

# New steering interactions
df['steering_x_sst']  = df['fl_steering'] * df['atlantic_sst']
df['upper_x_shear']   = df['upper_flow'] * df['wind_shear']
df['ridge_x_amo']     = df['ridge_pos'] * df['amo']
df['fav_x_steering']  = df['fav_index'] * (-df['fl_steering'])

df = df.dropna().reset_index(drop=True)
print(f"Dataset: {len(df)} years after all features")

FEATURES = [
    # Climate
    'atlantic_sst','gulf_sst','wind_shear','enso','amo',
    'orlando_rain','orlando_temp','year',
    # Lag
    'prev1','prev2','prev3',
    'prev_3yr_avg','prev_5yr_avg','prev_10yr_avg',
    # Anomalies + interactions
    'sst_anom','gulf_anom','rain_anom',
    'sst_x_amo','sst_x_enso','gulf_x_amo','shear_x_enso',
    'fav_index','post_1995','strong_lanina','amo_positive',
    'sst_trend_5yr',
    # NEW: Steering features
    'bermuda_high','fl_steering','upper_flow','ridge_pos',
    'steering_x_sst','upper_x_shear','ridge_x_amo','fav_x_steering',
]

# Check all correlations now
corr = df[FEATURES+['hurricane_count']].corr()['hurricane_count'].drop('hurricane_count')
print("\nTop 15 correlations with hurricane_count:")
print(corr.abs().sort_values(ascending=False).head(15).round(3).to_string())

# ── Split ──────────────────────────────────────────
train = df[df.year <= 2014]
val   = df[df.year >= 2015]

X_train = train[FEATURES].values
y_train = train['hurricane_count'].values
X_val   = val[FEATURES].values
y_val   = val['hurricane_count'].values

print(f"\nTrain: {len(train)} | Val: {len(val)}")
print(f"Val actual: {y_val.tolist()}")

# ── Models ─────────────────────────────────────────
models = {
    'XGB_v1': xgb.XGBRegressor(
        objective='count:poisson', max_depth=3, learning_rate=0.01,
        n_estimators=500, subsample=0.8, colsample_bytree=0.7,
        min_child_weight=1, random_state=42, verbosity=0),
    'XGB_v2': xgb.XGBRegressor(
        objective='count:poisson', max_depth=4, learning_rate=0.005,
        n_estimators=800, subsample=0.7, colsample_bytree=0.6,
        min_child_weight=2, random_state=42, verbosity=0),
    'RF': RandomForestRegressor(
        n_estimators=500, max_depth=5, min_samples_leaf=2,
        max_features=0.6, random_state=42),
    'GBM': GradientBoostingRegressor(
        n_estimators=300, max_depth=3, learning_rate=0.02,
        subsample=0.8, random_state=42),
    'ExtraTrees': ExtraTreesRegressor(
        n_estimators=500, max_depth=4, min_samples_leaf=2,
        random_state=42),
}

print("\n" + "="*72)
print(f"{'Model':<12} {'Within±1':>9} {'Exact':>7} {'MAE':>6}  Predictions")
print("="*72)

results = {}
for name, model in models.items():
    m = copy.deepcopy(model)
    m.fit(X_train, y_train)
    raw   = m.predict(X_val)
    preds = np.round(raw).clip(0).astype(int)

    within_1 = np.mean(np.abs(preds - y_val) <= 1) * 100
    exact    = np.mean(preds == y_val) * 100
    mae      = mean_absolute_error(y_val, preds)

    status = "✅" if within_1 >= 70 else "❌"
    print(f"{status} {name:<11} {within_1:>8.1f}% {exact:>6.1f}% "
          f"{mae:>6.2f}  raw:{np.round(raw,1).tolist()}")
    results[name] = {'model': m, 'within_1': within_1,
                     'preds': preds, 'mae': mae, 'raw': raw}

print("="*72)
print(f"  Actual:                          {y_val.tolist()}")

# ── Best model ─────────────────────────────────────
best_name = max(results, key=lambda k: (results[k]['within_1'],
                                         -results[k]['mae']))
best = results[best_name]

print(f"\n✅ Best: {best_name} — {best['within_1']:.1f}% Within±1")
print(f"\n── Year-by-Year ─────────────────────────────────")
detail = pd.DataFrame({
    'year':      val.year.values,
    'actual':    y_val,
    'predicted': best['preds'],
    'raw':       np.round(best['raw'], 2),
    'error':     np.abs(best['preds'] - y_val),
    'status':    ['✅' if e<=1 else '❌' for e in np.abs(best['preds'] - y_val)]
})
print(detail.to_string(index=False))

# Feature importance
try:
    imp = pd.DataFrame({
        'feature':    FEATURES,
        'importance': best['model'].feature_importances_
    }).sort_values('importance', ascending=False)
    print(f"\n── Top 10 Features ({best_name}) ────────────────")
    print(imp.head(10).to_string(index=False))
except:
    pass

# ── Save updated dataset ───────────────────────────
steering_cols = ['year','bermuda_high','fl_steering','upper_flow','ridge_pos']
df[['year','hurricane_count'] + FEATURES].to_csv(
    '/content/florida_hurricane_training_data_v2.csv', index=False)
print("\n✅ Saved: florida_hurricane_training_data_v2.csv")
